function deriveDigitalTwin(customer: any) {
  const category = String(customer?.favoriteCategory || customer?.favorite_category || 'Running');
  const productMap: Record<string, string> = {
    Running: 'Running Socks',
    Footwear: 'Running Socks',
    Activewear: 'Performance Tee',
    Hydration: 'Hydration Bottle',
    Accessories: 'Premium Cap',
  };
  const avgOrder = Number(customer?.avgOrderValue ?? customer?.avg_order_value ?? 2400);
  const frequency = Number(customer?.purchaseFrequency ?? customer?.purchase_frequency ?? 3);
  const score = Math.min(96, Math.max(71, 72 + frequency * 4 + (avgOrder > 3000 ? 9 : 5)));
  return {
    nextPredictedProduct: productMap[category] || 'Running Socks',
    confidenceScore: Math.round(score),
  };
